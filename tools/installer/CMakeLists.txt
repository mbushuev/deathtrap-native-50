cmake_minimum_required(VERSION 3.21)
project(native50_installer LANGUAGES CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use Visual Studio 2022 and -A Win32")
endif()
if(NOT EXISTS "${PAYLOAD_FILE}")
  message(FATAL_ERROR "PAYLOAD_FILE must identify the reviewed installer payload")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
configure_file(setup.rc.in "${CMAKE_CURRENT_BINARY_DIR}/setup.rc" @ONLY)
add_executable(Native50Setup WIN32 setup.cc core.cc cabinet.cc "${CMAKE_CURRENT_BINARY_DIR}/setup.rc")
target_compile_features(Native50Setup PRIVATE cxx_std_17)
target_compile_definitions(Native50Setup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
  WINVER=0x0601 _WIN32_WINNT=0x0601 NATIVE50_VERSION="${NATIVE50_VERSION}")
target_compile_options(Native50Setup PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_options(Native50Setup PRIVATE /SUBSYSTEM:WINDOWS,6.01 /DEBUG:NONE /MANIFEST:NO)
target_link_libraries(Native50Setup PRIVATE comctl32 shell32 ole32 advapi32 user32 gdi32 cabinet)
add_executable(installer_core_test core_test.cc core.cc cabinet.cc "${CMAKE_CURRENT_BINARY_DIR}/setup.rc")
target_compile_features(installer_core_test PRIVATE cxx_std_17)
target_compile_definitions(installer_core_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX
  WINVER=0x0601 _WIN32_WINNT=0x0601 NATIVE50_VERSION="${NATIVE50_VERSION}")
target_compile_options(installer_core_test PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(installer_core_test PRIVATE advapi32 user32 cabinet)
target_link_options(installer_core_test PRIVATE /MANIFEST:NO)
enable_testing()
add_test(NAME installer_rollback COMMAND installer_core_test)
